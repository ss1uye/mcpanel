  downloadable(instanceId: string, backupId: string): { path: string; filename: string } {
    const row = this.deps.db
      .select()
      .from(backups)
      .where(and(eq(backups.id, backupId), eq(backups.instanceId, instanceId)))
      .get();
    if (!row) throw new AppError('NOT_FOUND', '备份不存在');
    if (row.status !== 'ready') throw new AppError('INVALID_STATE', '备份尚不可下载');
    const file = path.join(this.deps.dataDir, 'backups', instanceId, row.filename);
    try {
      statSync(file);
    } catch {
      throw new AppError('NOT_FOUND', '备份文件不存在');
    }
    return { path: file, filename: row.filename };
  }

`apps/server/src/services/panel-backup-service.ts`：

```ts
import { randomBytes } from 'node:crypto';
import fs from 'node:fs/promises';
import path from 'node:path';
import type Database from 'better-sqlite3';
import { stamp } from '../utils/stamp';

export interface PanelBackupServiceDeps {
  sqlite: Database.Database;
  dataDir: string;
}

export interface PanelSnapshot {
  path: string;
  filename: string;
  sizeBytes: number;
}

export class PanelBackupService {
  constructor(private readonly deps: PanelBackupServiceDeps) {}

  async createSnapshot(): Promise<PanelSnapshot> {
    const dir = path.join(this.deps.dataDir, 'tmp');
    await fs.mkdir(dir, { recursive: true });
    const base = `panel-${stamp(new Date())}`;
    let filename = `${base}.db`;
    let target = path.join(dir, filename);
    for (let attempt = 0; attempt < 5; attempt += 1) {
      const exists = await fs
        .access(target)
        .then(() => true)
        .catch(() => false);
      if (!exists) break;
      filename = `${base}-${randomBytes(2).toString('hex')}.db`;
      target = path.join(dir, filename);
    }
    try {
      await this.deps.sqlite.backup(target);
    } catch (error) {
      await fs.rm(target, { force: true }).catch(() => undefined);
      throw error;
    }
    const stat = await fs.stat(target);
    return { path: target, filename, sizeBytes: stat.size };
  }
}
```

import { createReadStream } from 'node:fs';
import fs from 'node:fs/promises';
import type { FastifyInstance } from 'fastify';
import { panelSettingsPatchSchema } from '@mcpanel/shared';
import { createGuards } from '../auth/guard';
import type { SessionService } from '../auth/session';
import type { AuditService } from '../services/audit-service';
import type { PanelBackupService } from '../services/panel-backup-service';
import type { SettingsService } from '../services/settings-service';

export function registerSettingsRoutes(
  app: FastifyInstance,
  deps: {
    sessions: SessionService;
    audit: AuditService;
    settings: SettingsService;
    panelBackup?: PanelBackupService;
  },
): void {
  const { sessions, audit, settings, panelBackup } = deps;
  const { requireAuth, requireOrigin } = createGuards(sessions);

  app.get('/api/settings', { preHandler: [requireAuth] }, async () => settings.get());

  app.patch('/api/settings', { preHandler: [requireOrigin, requireAuth, app.csrfProtection] }, async (request) => {
    const patch = panelSettingsPatchSchema.parse(request.body);
    const next = settings.update(patch);
    audit.record({
      userId: request.user?.id,
      action: 'settings.update',
      targetType: 'settings',
      targetId: null,
      detail: { fields: Object.keys(patch) },
      ip: request.ip,
    });
    return next;
  });

  if (panelBackup) {
    app.get('/api/settings/panel-backup', { preHandler: [requireAuth] }, async (request, reply) => {
      const snapshot = await panelBackup.createSnapshot();
      audit.record({
        userId: request.user?.id,
        action: 'panel.backup',
        targetType: 'panel',
        targetId: null,
        detail: { filename: snapshot.filename, sizeBytes: snapshot.sizeBytes },
        ip: request.ip,
      });
      const stream = createReadStream(snapshot.path);
      const cleanup = () => {

  'X-Content-Type-Options': 'nosniff',
  'X-Frame-Options': 'DENY',
  'Referrer-Policy': 'no-referrer',
};

const HOST_PATTERN = /^(?:[A-Za-z0-9.-]+|\[[0-9A-Fa-f:.]+\])(?::\d{1,5})?$/;

export function buildCsp(host: string | undefined, nonce: string): string {
  const connect = ["'self'"];
  if (host && HOST_PATTERN.test(host)) connect.push(`ws://${host}`, `wss://${host}`);
  return [
    "default-src 'self'",
    "script-src 'self'",
    `style-src 'self' 'nonce-${nonce}'`,
    "img-src 'self' data:",
    "font-src 'self'",
    `connect-src ${connect.join(' ')}`,
    "frame-ancestors 'none'",
    "base-uri 'none'",
    "form-action 'none'",
  ].join('; ');
}

export function renderIndex(template: string, nonce: string): string {
  const replaced = template.replaceAll('__CSP_NONCE__', nonce);
  if (replaced.includes('name="csp-nonce"')) return replaced;
  return replaced.includes('</head>')
    ? replaced.replace('</head>', `<meta name="csp-nonce" content="${nonce}" /></head>`)
    : replaced;
}

export type EditorLanguage = 'json' | 'yaml' | 'properties';

export function detectEditorLanguage(path: string): EditorLanguage | null {
  const name = path.split(/[\\/]/).pop() ?? '';
  const dot = name.lastIndexOf('.');
  if (dot <= 0) return null;
  const extension = name.slice(dot + 1).toLowerCase();
  if (extension === 'json') return 'json';
  if (extension === 'yml' || extension === 'yaml') return 'yaml';
  if (extension === 'properties') return 'properties';
  return null;
}

import { EditorView } from '@codemirror/view';
import { basicSetup } from 'codemirror';
import { useEffect, useRef } from 'react';
import { readCspNonce } from '../utils/csp';
import { detectEditorLanguage } from '../utils/editor-language';

const theme = EditorView.theme(
  {
    '&': { height: '100%', backgroundColor: 'var(--console-bg)', color: 'var(--console-fg)', fontSize: '12px' },
    '.cm-scroller': { fontFamily: 'ui-monospace, Consolas, "Courier New", monospace', overflow: 'auto' },
    '.cm-gutters': { backgroundColor: 'var(--console-bg)', color: 'var(--muted)', border: 'none' },
    '.cm-activeLine': { backgroundColor: 'color-mix(in srgb, var(--surface-2) 55%, transparent)' },
    '.cm-activeLineGutter': { backgroundColor: 'transparent' },
    '&.cm-focused': { outline: 'none' },
  },
  { dark: true },
);

function languageExtension(path: string): Extension {
  const language = detectEditorLanguage(path);
  if (language === 'json') return json();
  if (language === 'yaml') return yaml();
  if (language === 'properties') return StreamLanguage.define(properties);
  return [];
}

export default function CodeEditor({
  path,
  value,
  onChange,
  ariaLabel,
  heightClass = 'h-80',
}: {
  path: string;
  value: string;
  onChange: (value: string) => void;
  ariaLabel: string;
  heightClass?: string;
}) {
  const host = useRef<HTMLDivElement>(null);
  const view = useRef<EditorView | null>(null);
  const changeRef = useRef(onChange);
  changeRef.current = onChange;

  useEffect(() => {
    if (!host.current) return undefined;
    const nonce = readCspNonce();
    const editor = new EditorView({
      parent: host.current,
      state: EditorState.create({
        doc: value,
        extensions: [
          basicSetup,
          oneDark,
          theme,
          EditorView.lineWrapping,
          languageExtension(path),
          EditorView.contentAttributes.of({ 'aria-label': ariaLabel }),
          EditorView.updateListener.of((update) => {
            if (update.docChanged) changeRef.current(update.state.doc.toString());
          }),
          ...(nonce ? [EditorView.cspNonce.of(nonce)] : []),
        ],
      }),
    });
    view.current = editor;
    return () => {
      editor.destroy();
      view.current = null;
    };
    // 只在切换文件时重建编辑器；value 的后续变化由下一个 effect 同步。
  }, [path]);

  useEffect(() => {
    const editor = view.current;
    if (!editor || editor.state.doc.toString() === value) return;
    editor.dispatch({ changes: { from: 0, to: editor.state.doc.length, insert: value } });
  }, [value]);

  return <div ref={host} className={`${heightClass} overflow-hidden rounded-lg border border-[var(--border)]`} />;
}
